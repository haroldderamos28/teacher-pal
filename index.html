function corsHeaders(request) { const origin = request.headers.get('Origin') || ''; const h = { 'Access-Control-Allow-Methods': 'POST, OPTIONS', 'Access-Control-Allow-Headers': 'Content-Type', 'Vary': 'Origin', }; if (ALLOWED_ORIGINS.includes(origin)) h['Access-Control-Allow-Origin'] = origin; return h; } const str = (v) => (typeof v === 'string' ? v.trim() : ''); function parseJson(raw) { if (raw && typeof raw === 'object') return raw; const s = String(raw || ''); const start = s.indexOf('{'); if (start === -1) return null; // Walk the string from the first "{" and find where that SAME object closes, // ignoring braces that appear inside quoted strings. This copes with models // that add extra text or stray braces after the real JSON object. let depth = 0, inStr = false, escape = false; for (let i = start; i < s.length; i++) { const c = s[i]; if (inStr) { if (escape) escape = false; else if (c === '\\') escape = true; else if (c === '"') inStr = false; continue; } if (c === '"') { inStr = true; continue; } if (c === '{') depth++; else if (c === '}') { depth--; if (depth === 0) { try { return JSON.parse(s.slice(start, i + 1)); } catch { return null; } } } } return null; } function buildPrompt(mode, level, extras) { const natural = MODES[mode]; const keys = [ '"correction": the sentence with grammar mistakes fixed, changing as little as possible. Keep the meaning.'); keys.push('"explanation": one short sentence (max 15 words) explaining the most important correction. Empty string if nothing needed correcting.'); keys.push('"followup": one short, natural question the teacher could ask next to keep the conversation going, or an empty string.'); if (extras) { keys.push('"alternatives": an array of 2 other natural ways to say the same idea.'); keys.push('"vocabulary": one stronger word or phrase that fits this context, with a tiny example, or an empty string.'); } return ( 'You assist an English teacher. You receive one sentence spoken by an English learner, from speech transcription, ' + 'so ignore missing punctuation or capital letters. ' + (LEVELS[level] || LEVELS.intermediate) + ' ' + 'Reply with ONLY a JSON object with these keys:\n' + keys.join('\n') + '\n' + 'Never invent meaning the learner did not express. Be brief. Use exactly these keys, no others, and no nested objects — every value is a plain string or array of strings. No text outside the JSON.' ); } function buildTranslatePrompt(langCode, level) { const lang = LANGUAGE_NAMES[langCode] || 'the student\'s language'; return ( 'You assist an English teacher whose student is a beginner English learner. The student just said something ' + 'in ' + lang + ' (from speech transcription, so ignore missing punctuation). ' + (LEVELS[level] || LEVELS.basic) + ' ' + 'Reply with ONLY a JSON object with these keys:\n' + '"translation": a natural, accurate English translation of exactly what they said. Do not add anything they did not say.\n' + '"suggestion": one short, simple English sentence the student could learn to say to express this themselves, matching their level, or an empty string if none fits.\n' + 'Be brief. Use exactly these keys, no others, no nested objects. No text outside the JSON.' ); } export default { async fetch(request, env) { const cors = corsHeaders(request); const json = (obj, status = 200) => new Response(JSON.stringify(obj), { status, headers: { 'Content-Type': 'application/json', ...cors }, }); if (request.method === 'OPTIONS') return new Response(null, { status: 204, headers: cors }); if (request.method !== 'POST') return json({ error: 'Use POST' }, 405); const path = new URL(request.url).pathname; // ---- Deepgram temporary token ---- if (path === '/token') { if (!env.DEEPGRAM_API_KEY) return json({ error: 'DEEPGRAM_API_KEY secret is missing' }, 500); try { const r = await fetch('https://api.deepgram.com/v1/auth/grant', { method: 'POST', headers: { Authorization: 'Token ' + env.DEEPGRAM_API_KEY, 'Content-Type': 'application/json', }, }); const data = await r.json(); if (!r.ok || !data.access_token) { return json({ error: 'Deepgram refused: ' + (data.err_msg || data.message || r.status) }, 502); } return json({ token: data.access_token }); } catch (e) { return json({ error: 'Could not reach Deepgram' }, 502); } } // ---- Analyse or translate one sentence ---- let body; try { body = await request.json(); } catch { return json({ error: 'Bad JSON' }, 400); } const text = String(body.text || '').trim().slice(0, 500); if (!text) return json({ error: 'No text' }, 400); const level = body.level in LEVELS ? body.level : 'intermediate'; const translate = body.translate === true; let messages, maxTokens; if (translate) { const langCode = String(body.lang || '').trim(); messages = [ { role: 'system', content: buildTranslatePrompt(langCode, level) }, { role: 'user', content: text }, ]; maxTokens = 200; } else { const mode = body.mode in MODES ? body.mode : 'natural'; const extras = body.extras === true; messages = [ { role: 'system', content: buildPrompt(mode, level, extras) }, { role: 'user', content: text }, ]; // Scale the reply budget with how long the sentence is, so a longer sentence // (e.g. one with no internal punctuation for the client to split on) still gets // enough room to come back as valid JSON instead of being cut off mid-reply. const base = extras ? 300 : 170; maxTokens = Math.min(500, base + Math.max(0, text.split(/\s+/).length - 12) * 6); } let lastError = 'No model worked'; for (const model of MODELS) { try { const r = await env.AI.run(model, { messages, max_tokens: maxTokens }); const raw = r.response ?? r.choices?.[0]?.message?.content ??
